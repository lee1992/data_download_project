````
"""
Phase 0 — 数据工程
==================
负责：
  - 读取并标准化 5-min OHLCV CSV
  - 数据质量校验
  - 多频率重采样（从5min向上聚合，跨日不混合）
  - 交易时段标记与缺口标记

PyCharm Console 调试示例：
    from ew_system.phase0_data import DataLoader
    loader = DataLoader("futu_data/K_5M/US_QQQ_K_5M_2026-04-22_2026-04-24.csv")
    df = loader.load()
    print(df.head())
    print(loader.issues)
    df_1h = loader.resample("1h")
    print(df_1h)
"""

from __future__ import annotations

import warnings
from dataclasses import dataclass, field
from pathlib import Path
from typing import Dict, List, Optional

import numpy as np
import pandas as pd


# ---------------------------------------------------------------------------
# 数据质量报告
# ---------------------------------------------------------------------------

@dataclass
class ValidationReport:
    total_bars: int = 0
    duplicates_removed: int = 0
    nan_rows: int = 0
    ohlc_violations: int = 0
    time_gaps: List[tuple] = field(default_factory=list)
    is_valid: bool = True
    messages: List[str] = field(default_factory=list)

    def print(self):
        print(f"  总K线数: {self.total_bars}")
        for msg in self.messages:
            print(f"  {msg}")


# ---------------------------------------------------------------------------
# 主类
# ---------------------------------------------------------------------------

class DataLoader:
    """
    读取、验证并重采样 5-min OHLCV CSV。

    支持的文件格式（你的富途数据）：
        code, name, time_key, open, close, high, low, ..., volume, ...

    参数
    ----
    path : str | Path
        CSV 文件路径
    tz : str | None
        时区字符串（如 'US/Eastern', 'Asia/Shanghai'）。
        None 则根据 code 列自动推断。
    """

    # 列名映射：CSV列名 → 内部标准列名
    _COL_MAP = {
        "time_key": "datetime",
        "open":     "open",
        "high":     "high",
        "low":      "low",
        "close":    "close",
        "volume":   "volume",
        "code":     "code",
    }

    def __init__(self, path: str | Path, tz: str | None = None):
        self.path = Path(path)
        self.df: pd.DataFrame = pd.DataFrame()
        self.issues: List[str] = []
        self.report: Optional[ValidationReport] = None
        self._tz = tz

    # ------------------------------------------------------------------ #
    #  公开 API
    # ------------------------------------------------------------------ #

    def load(self) -> pd.DataFrame:
        """
        完整加载流程：读取 → 标准化 → 推断时区 → 校验 → 标记会话。
        返回干净的 DataFrame。
        """
        self._read_csv()
        self._standardise()
        self._infer_timezone()
        self._validate()
        self._mark_sessions()
        return self.df

    def resample(self, rule: str) -> pd.DataFrame:
        """
        将已加载的 5-min 数据重采样到更粗粒度。

        支持的 rule：
            '15min', '30min', '1h', '4h', 'D', 'W', 'M'

        日内频率（15min/30min/1h/4h）：按交易日分组重采样，
        避免跨日缺口混入同一根K线。

        日线及以上（D/W/M）：按日期/周/月聚合。
        """
        if self.df.empty:
            raise RuntimeError("请先调用 .load()")
        return self._resample_impl(rule)

    def build_multi_tf(
        self,
        timeframes: List[str] | None = None
    ) -> Dict[str, pd.DataFrame]:
        """
        一次性构建多频率字典。

        返回：{'5min': df, '15min': df, '1h': df, ...}
        """
        if timeframes is None:
            timeframes = ["15min", "30min", "1h", "4h", "D"]
        result = {"5min": self.df.copy()}
        for tf in timeframes:
            result[tf] = self.resample(tf)
        return result

    # ------------------------------------------------------------------ #
    #  内部方法
    # ------------------------------------------------------------------ #

    def _read_csv(self):
        self.df = pd.read_csv(self.path, encoding="utf-8-sig")

    def _standardise(self):
        df = self.df.copy()

        # 重命名
        rename = {k: v for k, v in self._COL_MAP.items() if k in df.columns}
        df.rename(columns=rename, inplace=True)

        # 解析时间
        df["datetime"] = pd.to_datetime(df["datetime"])
        df.sort_values("datetime", inplace=True)
        df.reset_index(drop=True, inplace=True)

        # 只保留需要的列
        keep = ["datetime", "open", "high", "low", "close", "volume"]
        if "code" in df.columns:
            keep.insert(0, "code")
        df = df[[c for c in keep if c in df.columns]].copy()

        # 数值类型
        for c in ["open", "high", "low", "close", "volume"]:
            if c in df.columns:
                df[c] = pd.to_numeric(df[c], errors="coerce")

        self.df = df

    def _infer_timezone(self):
        if self._tz:
            return
        code = ""
        if "code" in self.df.columns and len(self.df) > 0:
            code = str(self.df["code"].iloc[0])
        if any(code.startswith(p) for p in ("US.", "NASDAQ", "NYSE", "AMEX")):
            self._tz = "US/Eastern"
        else:
            self._tz = "Asia/Shanghai"

    def _validate(self) -> ValidationReport:
        df = self.df
        rpt = ValidationReport(total_bars=len(df))

        # 1) 重复时间戳
        dups = df["datetime"].duplicated()
        if dups.any():
            rpt.duplicates_removed = int(dups.sum())
            rpt.messages.append(f"⚠ 重复时间戳: {rpt.duplicates_removed} 行（已删除）")
            df = df[~dups].copy()

        # 2) OHLC 逻辑
        bad_hl = df["high"] < df["low"]
        if bad_hl.any():
            rpt.ohlc_violations += int(bad_hl.sum())
            rpt.messages.append(f"⚠ high < low: {bad_hl.sum()} 行")

        # 3) NaN
        nulls = df[["open", "high", "low", "close"]].isna().any(axis=1)
        if nulls.any():
            rpt.nan_rows = int(nulls.sum())
            rpt.messages.append(f"⚠ OHLC含NaN: {rpt.nan_rows} 行（已删除）")
            df.dropna(subset=["open", "high", "low", "close"], inplace=True)

        # 4) 日内时间缺口（跨日不报）
        if len(df) > 1:
            diffs = df["datetime"].diff().dt.total_seconds() / 60
            for i in range(1, len(diffs)):
                gap = diffs.iloc[i]
                if pd.notna(gap) and 10 < gap < 720:  # 10min~12h 视为异常缺口
                    rpt.time_gaps.append((i, gap))
            if rpt.time_gaps:
                rpt.messages.append(
                    f"⚠ 日内时间缺口: {len(rpt.time_gaps)} 处"
                )

        rpt.is_valid = (rpt.nan_rows == 0 and rpt.ohlc_violations == 0)
        if rpt.is_valid and not rpt.messages:
            rpt.messages.append("✔ 数据质量检查通过")

        self.issues = [m for m in rpt.messages if m.startswith("⚠")]
        self.report = rpt
        self.df = df.reset_index(drop=True)
        return rpt

    def _mark_sessions(self):
        """添加辅助列：trade_date, bar_of_day, is_gap, gap_pct。"""
        df = self.df
        df["trade_date"] = df["datetime"].dt.date
        df["bar_of_day"] = df.groupby("trade_date").cumcount()

        df["is_gap"] = False
        df["gap_pct"] = 0.0
        dates = df["trade_date"].unique()
        for i in range(1, len(dates)):
            prev_mask = df["trade_date"] == dates[i - 1]
            curr_mask = df["trade_date"] == dates[i]
            if prev_mask.any() and curr_mask.any():
                prev_close = df.loc[prev_mask, "close"].iloc[-1]
                curr_open = df.loc[curr_mask, "open"].iloc[0]
                gap_pct = (curr_open - prev_close) / prev_close * 100
                first_idx = df.index[curr_mask][0]
                df.loc[first_idx, "is_gap"] = True
                df.loc[first_idx, "gap_pct"] = round(gap_pct, 4)

        self.df = df

    def _resample_impl(self, rule: str) -> pd.DataFrame:
        """实际重采样逻辑。"""
        rule_upper = rule.upper()

        # 日线及以上
        if rule_upper in ("D", "W", "M"):
            df = self.df.copy()
            df["_dt"] = pd.to_datetime(df["trade_date"])
            if rule_upper == "D":
                grp_key = df["trade_date"]
            elif rule_upper == "W":
                grp_key = df["_dt"].dt.to_period("W").astype(str)
            else:
                grp_key = df["_dt"].dt.to_period("M").astype(str)

            agg = df.groupby(grp_key).agg(
                datetime=("datetime", "first"),
                open=("open", "first"),
                high=("high", "max"),
                low=("low", "min"),
                close=("close", "last"),
                volume=("volume", "sum"),
            ).reset_index(drop=True)
            return agg

        # 日内：按交易日分组，避免跨日混合
        # 将 rule 转换为 pandas offset alias
        alias_map = {
            "15MIN": "15min", "30MIN": "30min",
            "1H": "1h", "60MIN": "1h",
            "4H": "4h", "240MIN": "4h",
        }
        pd_rule = alias_map.get(rule_upper, rule)

        frames = []
        for date, gdf in self.df.groupby("trade_date"):
            tmp = gdf.set_index("datetime")[["open", "high", "low", "close", "volume"]]
            r = tmp.resample(pd_rule).agg({
                "open": "first", "high": "max",
                "low": "min", "close": "last",
                "volume": "sum",
            }).dropna(subset=["open"])
            frames.append(r)

        if not frames:
            return pd.DataFrame(columns=["datetime", "open", "high", "low", "close", "volume"])

        out = pd.concat(frames)
        out.index.name = "datetime"
        return out.reset_index()


# ---------------------------------------------------------------------------
# 便捷函数
# ---------------------------------------------------------------------------

def load_ohlcv(path: str | Path, tz: str | None = None) -> pd.DataFrame:
    """一行加载数据的便捷函数。"""
    loader = DataLoader(path, tz=tz)
    return loader.load()
````

````
"""
Phase 1 — 极值点识别
====================
负责：
  - 自适应 ZigZag（ATR 阈值）
  - 多尺度极值点（MultiScalePivots）
  - Pivot / Segment 数据结构
  - 嵌套一致性验证

核心参考：
  - Kaufman《Trading Systems and Methods》ZigZag 章节
  - Neely《Mastering Elliott Wave》极值点定义
  - ATR 自适应阈值：避免固定百分比在不同波动率下失效

PyCharm Console 调试示例：
    from ew_system.phase0_data import DataLoader
    from ew_system.phase1_pivots import PivotEngine, MultiScalePivots
    loader = DataLoader("futu_data/K_5M/US_QQQ_K_5M_2026-04-22_2026-04-24.csv")
    df = loader.load()
    engine = PivotEngine(atr_mult=1.5)
    pivots, atr = engine.find_pivots(df)
    for p in pivots:
        print(p)
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional, Tuple

import numpy as np
import pandas as pd


# ---------------------------------------------------------------------------
# 数据结构
# ---------------------------------------------------------------------------

@dataclass
class Pivot:
    """
    一个极值点（拐点）。

    属性
    ----
    idx       : 在源 DataFrame 中的行位置索引
    datetime  : 时间戳
    price     : 极值价格（swing high 用 high，swing low 用 low）
    kind      : 'H'（高点）或 'L'（低点）
    atr       : 该点的 ATR 值（用于后续比例计算）
    scale     : 所属尺度（ATR 倍数），MultiScalePivots 中使用
    is_anchor : 是否为用户指定的锚点
    anchor_role : 锚点的结构角色（如 'wave3_end'）
    """
    idx: int
    datetime: Any
    price: float
    kind: str                        # 'H' or 'L'
    atr: float = 0.0
    scale: float = 0.0
    is_anchor: bool = False
    anchor_role: Optional[str] = None

    def __repr__(self):
        anc = f" [ANCHOR:{self.anchor_role}]" if self.is_anchor else ""
        return (f"Pivot({self.kind} "
                f"idx={self.idx} "
                f"dt={pd.Timestamp(self.datetime).strftime('%Y-%m-%d %H:%M')} "
                f"px={self.price:.4f}{anc})")

    @property
    def ts(self) -> pd.Timestamp:
        return pd.Timestamp(self.datetime)


@dataclass
class Segment:
    """
    两个相邻 Pivot 之间的一段走势（波段）。

    用于 Phase 2/3 的比例计算和时间分析。
    """
    start: Pivot
    end: Pivot
    direction: int          # +1 上涨, -1 下跌
    price_move: float       # 有符号价格变动（end.price - start.price）
    pct_move: float         # 百分比变动
    duration_bars: int      # 持续 bar 数
    duration_min: float     # 持续分钟数

    @property
    def length(self) -> float:
        """绝对价格变动幅度。"""
        return abs(self.price_move)

    @property
    def length_pct(self) -> float:
        """绝对百分比变动。"""
        return abs(self.pct_move)

    def __repr__(self):
        d = "↑" if self.direction == 1 else "↓"
        return (f"Seg({d} {self.pct_move:+.2f}% "
                f"{self.duration_bars}bars "
                f"{self.start.price:.4f}→{self.end.price:.4f})")


# ---------------------------------------------------------------------------
# ATR 计算
# ---------------------------------------------------------------------------

def compute_atr(df: pd.DataFrame, period: int = 14) -> pd.Series:
    """
    计算 Average True Range。

    True Range = max(H-L, |H-prev_C|, |L-prev_C|)
    返回与 df 等长的 Series（前 period-1 个值用扩展窗口）。
    """
    h = df["high"].values
    l = df["low"].values
    c = df["close"].values
    n = len(df)

    tr = np.empty(n)
    tr[0] = h[0] - l[0]
    for i in range(1, n):
        tr[i] = max(
            h[i] - l[i],
            abs(h[i] - c[i - 1]),
            abs(l[i] - c[i - 1]),
        )

    atr = pd.Series(tr, index=df.index).rolling(period, min_periods=1).mean()
    return atr


# ---------------------------------------------------------------------------
# 核心 ZigZag 引擎
# ---------------------------------------------------------------------------

class PivotEngine:
    """
    自适应 ZigZag 极值点识别引擎。

    算法思路
    --------
    1. 计算 ATR，用 ATR × atr_mult 作为动态阈值
    2. 状态机扫描：undecided → looking_for_high / looking_for_low
    3. 当价格从极值回撤超过阈值时，确认该极值为 Pivot
    4. 强制高低交替（相邻同类取更极端值）

    参数
    ----
    atr_period : int
        ATR 计算周期（默认 14 根 K 线 = 70 分钟 on 5min 数据）
    atr_mult : float
        ATR 倍数阈值。
        较小（1.0-1.5）→ 更多 pivot → 捕捉小浪
        较大（3.0-5.0）→ 更少 pivot → 捕捉大浪
    price_source : str
        'hl'    → swing high 用 high，swing low 用 low（推荐）
        'close' → 全部用 close
    hard_pivot_times : list
        必须出现为 pivot 的时间戳列表（Phase-1 锚点）
    hard_pivot_window : int
        硬锚点时间匹配窗口（bar 数）
    """

    def __init__(
        self,
        atr_period: int = 14,
        atr_mult: float = 2.0,
        price_source: str = "hl",
        hard_pivot_times: list | None = None,
        hard_pivot_window: int = 3,
    ):
        self.atr_period = atr_period
        self.atr_mult = atr_mult
        self.price_source = price_source
        self.hard_pivot_times = hard_pivot_times or []
        self.hard_pivot_window = hard_pivot_window

    def find_pivots(
        self, df: pd.DataFrame
    ) -> Tuple[List[Pivot], pd.Series]:
        """
        主入口：返回 (pivots, atr_series)。

        df 必须包含列：datetime, open, high, low, close
        """
        atr = compute_atr(df, self.atr_period)
        pivots = self._zigzag(df, atr)

        if self.hard_pivot_times:
            pivots = self._inject_hard_pivots(df, atr, pivots)

        return pivots, atr

    def pivots_to_segments(self, pivots: List[Pivot]) -> List[Segment]:
        """将相邻 Pivot 转换为 Segment 列表。"""
        segs = []
        for i in range(len(pivots) - 1):
            a, b = pivots[i], pivots[i + 1]
            move = b.price - a.price
            direction = 1 if move > 0 else -1
            pct = move / a.price * 100 if a.price else 0.0
            dur_bars = b.idx - a.idx
            try:
                dur_min = (b.ts - a.ts).total_seconds() / 60
            except Exception:
                dur_min = dur_bars * 5.0  # fallback: 5min per bar
            segs.append(Segment(a, b, direction, move, pct, dur_bars, dur_min))
        return segs

    # ------------------------------------------------------------------ #
    #  内部方法
    # ------------------------------------------------------------------ #

    def _zigzag(self, df: pd.DataFrame, atr: pd.Series) -> List[Pivot]:
        """核心自适应 ZigZag 状态机。"""
        n = len(df)
        if n < 3:
            return []

        highs = df["high"].values
        lows = df["low"].values
        closes = df["close"].values
        dts = df["datetime"].values
        atr_vals = atr.values
        use_hl = (self.price_source == "hl")

        pivots: List[Pivot] = []
        direction = 0  # 0=未定, 1=找高点, -1=找低点

        last_high_idx = 0
        last_high_px = highs[0] if use_hl else closes[0]
        last_low_idx = 0
        last_low_px = lows[0] if use_hl else closes[0]

        def make_pivot(idx: int, kind: str) -> Pivot:
            px = (highs[idx] if kind == "H" else lows[idx]) if use_hl else closes[idx]
            return Pivot(
                idx=idx,
                datetime=dts[idx],
                price=px,
                kind=kind,
                atr=float(atr_vals[idx]),
            )

        for i in range(1, n):
            hi = highs[i] if use_hl else closes[i]
            lo = lows[i] if use_hl else closes[i]
            thr = max(atr_vals[i] * self.atr_mult, 1e-8)

            if direction == 0:
                if hi - last_low_px >= thr:
                    pivots.append(make_pivot(last_low_idx, "L"))
                    direction = 1
                    last_high_idx, last_high_px = i, hi
                elif last_high_px - lo >= thr:
                    pivots.append(make_pivot(last_high_idx, "H"))
                    direction = -1
                    last_low_idx, last_low_px = i, lo
                else:
                    if hi > last_high_px:
                        last_high_idx, last_high_px = i, hi
                    if lo < last_low_px:
                        last_low_idx, last_low_px = i, lo

            elif direction == 1:
                if hi > last_high_px:
                    last_high_idx, last_high_px = i, hi
                elif last_high_px - lo >= thr:
                    pivots.append(make_pivot(last_high_idx, "H"))
                    direction = -1
                    last_low_idx, last_low_px = i, lo

            else:  # direction == -1
                if lo < last_low_px:
                    last_low_idx, last_low_px = i, lo
                elif hi - last_low_px >= thr:
                    pivots.append(make_pivot(last_low_idx, "L"))
                    direction = 1
                    last_high_idx, last_high_px = i, hi

        # 收尾：添加最后一个未确认的极值
        if direction == 1:
            pivots.append(make_pivot(last_high_idx, "H"))
        elif direction == -1:
            pivots.append(make_pivot(last_low_idx, "L"))

        return self._enforce_alternation(pivots)

    @staticmethod
    def _enforce_alternation(pivots: List[Pivot]) -> List[Pivot]:
        """
        强制高低交替：相邻同类 pivot 保留更极端的那个。
        这是 Elliott Wave 分析的基本前提。
        """
        if len(pivots) < 2:
            return pivots
        clean = [pivots[0]]
        for p in pivots[1:]:
            if p.kind == clean[-1].kind:
                if p.kind == "H":
                    if p.price >= clean[-1].price:
                        clean[-1] = p
                else:
                    if p.price <= clean[-1].price:
                        clean[-1] = p
            else:
                clean.append(p)
        return clean

    def _inject_hard_pivots(
        self,
        df: pd.DataFrame,
        atr: pd.Series,
        pivots: List[Pivot],
    ) -> List[Pivot]:
        """
        将用户指定的硬锚点时间强制插入 pivot 列表。
        如果该时间附近已有 pivot，则标记为 anchor；
        否则创建新 pivot 并插入。
        """
        dts = df["datetime"].values
        existing_idxs = {p.idx for p in pivots}

        for ts_raw in self.hard_pivot_times:
            ts = pd.Timestamp(ts_raw)
            # 找最近的 bar
            diffs = np.abs(pd.to_datetime(dts) - ts)
            nearest_bar = int(np.argmin(diffs))

            # 在窗口内查找已有 pivot
            found = None
            for p in pivots:
                if abs(p.idx - nearest_bar) <= self.hard_pivot_window:
                    found = p
                    break

            if found is not None:
                found.is_anchor = True
            else:
                # 创建新 pivot
                hi = df["high"].iloc[nearest_bar]
                lo = df["low"].iloc[nearest_bar]
                cl = df["close"].iloc[nearest_bar]
                kind = "H" if cl >= (hi + lo) / 2 else "L"
                px = hi if kind == "H" else lo
                new_p = Pivot(
                    idx=nearest_bar,
                    datetime=dts[nearest_bar],
                    price=px,
                    kind=kind,
                    atr=float(atr.iloc[nearest_bar]),
                    is_anchor=True,
                )
                pivots.append(new_p)

        pivots.sort(key=lambda p: p.idx)
        return self._enforce_alternation(pivots)


# ---------------------------------------------------------------------------
# 多尺度极值点
# ---------------------------------------------------------------------------

class MultiScalePivots:
    """
    用不同 ATR 倍数生成多层级极值点集合。

    核心思想（来自 Neely NeoWave）：
      大尺度极值点 ⊂ 小尺度极值点（在时间容差内）
      这为 Phase 3 的嵌套分析提供自然的层级基础。

    参数
    ----
    df : pd.DataFrame
        标准化的 OHLCV 数据
    scales : list[float]
        ATR 倍数列表，从小到大排列
        建议：[1.0, 2.0, 3.5, 6.0, 10.0]
        对应：小浪 / 中浪 / 主浪 / 周期浪 / 超级周期浪
    atr_period : int
        ATR 计算周期
    merge_tolerance_bars : int
        嵌套验证时的时间容差（bar 数）

    使用示例
    --------
        msp = MultiScalePivots(df, scales=[1.0, 2.5, 5.0])
        print(msp.summary())
        pivots_small = msp.get_pivots(1.0)
        pivots_large = msp.get_pivots(5.0)
        issues = msp.verify_nesting()
    """

    def __init__(
        self,
        df: pd.DataFrame,
        scales: List[float] | None = None,
        atr_period: int = 14,
        merge_tolerance_bars: int = 3,
    ):
        if scales is None:
            scales = [1.0, 2.0, 3.5, 6.0, 10.0]

        self.df = df
        self.scales = sorted(scales)
        self.atr_period = atr_period
        self.merge_tolerance = merge_tolerance_bars

        # 为每个尺度生成 pivot 集合
        self.pivot_sets: Dict[float, List[Pivot]] = {}
        self._atr: Optional[pd.Series] = None

        for s in self.scales:
            engine = PivotEngine(atr_period=atr_period, atr_mult=s)
            pvts, atr = engine.find_pivots(df)
            if self._atr is None:
                self._atr = atr
            for p in pvts:
                p.scale = s
            self.pivot_sets[s] = pvts

    def get_pivots(self, scale: float) -> List[Pivot]:
        """获取指定尺度的 pivot 列表。"""
        return self.pivot_sets.get(scale, [])

    def get_segments(self, scale: float) -> List[Segment]:
        """获取指定尺度的 segment 列表。"""
        engine = PivotEngine()
        return engine.pivots_to_segments(self.get_pivots(scale))

    def verify_nesting(
        self, tolerance_bars: int | None = None
    ) -> List[str]:
        """
        验证大尺度极值点是否（在容差内）是小尺度极值点的子集。

        理论依据：Elliott Wave 的嵌套性要求大级别的转折点
        必须也是小级别的转折点。

        返回：问题列表（空列表 = 全部通过）
        """
        tol = tolerance_bars or self.merge_tolerance
        issues = []

        for i in range(len(self.scales) - 1):
            small_s = self.scales[i]
            large_s = self.scales[i + 1]
            small_idxs = {p.idx for p in self.pivot_sets[small_s]}
            large_pvts = self.pivot_sets[large_s]

            for lp in large_pvts:
                found = any(
                    abs(lp.idx - si) <= tol for si in small_idxs
                )
                if not found:
                    issues.append(
                        f"Scale {large_s}x pivot @ idx={lp.idx} "
                        f"({lp.ts.strftime('%m-%d %H:%M')}) "
                        f"不在 scale {small_s}x 的 pivot 集合中"
                    )

        return issues

    def summary(self) -> str:
        """打印各尺度 pivot 数量摘要。"""
        lines = ["MultiScalePivots 摘要:"]
        for s in self.scales:
            n = len(self.pivot_sets[s])
            lines.append(f"  Scale {s:>5.1f}x ATR : {n:>4d} pivots")
        return "\n".join(lines)
````

````
"""
锚点系统 (Anchor System)
========================
允许用户注入先验知识：
  - 某个时间点必须是 pivot（Hard Pivot）
  - 某个时间点必须是某浪的端点（Structural Anchor）
  - 某个时间点大概率是重要拐点（Soft Hint）

锚点在 Phase 1 和 Phase 2 中均有应用：
  Phase 1：Hard Pivot → 强制插入 pivot 集合
  Phase 2：Structural → 约束浪形搜索的节点位置

PyCharm Console 调试示例：
    from ew_system.anchors import AnchorManager, AnchorStrength
    from ew_system.phase1_pivots import PivotType
    mgr = AnchorManager()
    mgr.add_pivot_anchor("2026-04-23 13:50:00", "L", strength="hard", role="wave4_end")
    print(mgr.get_structural_constraints())
"""

from __future__ import annotations

import warnings
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any, Dict, List, Optional, Union

import numpy as np
import pandas as pd


# ---------------------------------------------------------------------------
# 锚点强度枚举
# ---------------------------------------------------------------------------

class AnchorStrength:
    SOFT       = "soft"        # 软提示：提高评分权重，不强制
    HARD_PIVOT = "hard"        # 硬锚点：必须是 pivot
    STRUCTURAL = "structural"  # 结构锚点：必须是某浪的端点


# ---------------------------------------------------------------------------
# 锚点数据类
# ---------------------------------------------------------------------------

@dataclass
class Anchor:
    """
    用户指定的先验锚点。

    参数
    ----
    datetime      : 时间点（字符串或 datetime）
    kind          : 'H'（高点）或 'L'（低点），None 表示不限
    strength      : AnchorStrength 中的一个值
    role          : 结构角色，如 'wave1_end', 'wave3_end', 'wave5_end',
                    'wave2_end', 'wave4_end', 'waveA_end', 'waveB_end', 'waveC_end'
    price         : 可选，用于验证匹配精度
    tolerance_bars: 时间匹配容差（bar 数）
    note          : 备注说明

    示例
    ----
        # 确定某低点是4浪终点
        Anchor("2026-04-23 13:50:00", kind="L",
               strength="structural", role="wave4_end")

        # 确定某高点是3浪终点
        Anchor("2026-04-23 11:30:00", kind="H",
               strength="structural", role="wave3_end")
    """
    datetime: Any
    kind: Optional[str] = None          # 'H' or 'L'
    strength: str = AnchorStrength.HARD_PIVOT
    role: Optional[str] = None          # 结构角色
    price: Optional[float] = None
    tolerance_bars: int = 5
    note: str = ""

    def __post_init__(self):
        self.datetime = pd.Timestamp(self.datetime)

    def __repr__(self):
        return (f"Anchor({self.strength} @ {self.datetime.strftime('%Y-%m-%d %H:%M')}, "
                f"kind={self.kind}, role={self.role})")


# ---------------------------------------------------------------------------
# 锚点管理器
# ---------------------------------------------------------------------------

class AnchorManager:
    """
    管理所有锚点，提供 Phase 1 和 Phase 2 的约束接口。
    """

    def __init__(self):
        self.anchors: List[Anchor] = []

    def add(self, anchor: Anchor):
        """添加一个锚点对象。"""
        self.anchors.append(anchor)

    def add_pivot_anchor(
        self,
        dt: Union[str, datetime],
        kind: Optional[str] = None,
        strength: str = AnchorStrength.HARD_PIVOT,
        role: Optional[str] = None,
        price: Optional[float] = None,
        tolerance_bars: int = 5,
        note: str = "",
    ):
        """
        便捷方法：添加一个锚点。

        参数
        ----
        dt            : 时间点
        kind          : 'H' 或 'L'
        strength      : 'soft', 'hard', 'structural'
        role          : 结构角色（structural 时必填）
        price         : 可选价格
        tolerance_bars: 时间容差
        note          : 备注
        """
        self.anchors.append(Anchor(
            datetime=dt,
            kind=kind,
            strength=strength,
            role=role,
            price=price,
            tolerance_bars=tolerance_bars,
            note=note,
        ))

    def get_hard_pivot_times(self) -> List[str]:
        """
        返回所有 hard/structural 锚点的时间戳列表。
        供 PivotEngine 的 hard_pivot_times 参数使用。
        """
        return [
            str(a.datetime)
            for a in self.anchors
            if a.strength in (AnchorStrength.HARD_PIVOT, AnchorStrength.STRUCTURAL)
        ]

    def get_structural_constraints(self) -> List[Dict]:
        """
        返回所有 structural 锚点的约束字典列表。
        供 Phase 2 的 WaveSearcher 使用。

        返回格式：
        [
            {
                'datetime': pd.Timestamp,
                'kind': 'H' or 'L',
                'role': 'wave3_end',
                'tolerance_bars': 5,
                'price': float or None,
            },
            ...
        ]
        """
        return [
            {
                "datetime": a.datetime,
                "kind": a.kind,
                "role": a.role,
                "tolerance_bars": a.tolerance_bars,
                "price": a.price,
            }
            for a in self.anchors
            if a.strength == AnchorStrength.STRUCTURAL and a.role
        ]

    def apply_to_pivots(
        self,
        pivots: list,
        df: pd.DataFrame,
    ) -> list:
        """
        将锚点标记应用到已有的 pivot 列表。
        匹配到的 pivot 会被标记 is_anchor=True 和 anchor_role。
        """
        for anchor in self.anchors:
            matched = self._find_matching_pivot(
                pivots, anchor.datetime, anchor.tolerance_bars
            )
            if matched is not None:
                matched.is_anchor = True
                matched.anchor_role = anchor.role
        return pivots

    def resolve_structural_to_pivot_indices(
        self,
        pivots: list,
    ) -> Dict[str, int]:
        """
        将 structural 约束解析为 pivot 列表中的索引。

        返回：{'wave3_end': pivot_list_index, ...}
        供 Phase 2 的 anchors 参数使用。
        """
        result = {}
        for constraint in self.get_structural_constraints():
            role = constraint["role"]
            matched_idx = self._find_matching_pivot_index(
                pivots,
                constraint["datetime"],
                constraint["tolerance_bars"],
            )
            if matched_idx is not None:
                # 将 role 映射到 wave slot
                slot = self._role_to_slot(role)
                if slot is not None:
                    result[slot] = matched_idx
        return result

    # ------------------------------------------------------------------ #
    #  内部方法
    # ------------------------------------------------------------------ #

    @staticmethod
    def _find_matching_pivot(pivots: list, ts: pd.Timestamp, tol: int):
        """在 pivot 列表中找到时间最近且在容差内的 pivot。"""
        best = None
        best_dist = None
        for p in pivots:
            d = abs((pd.Timestamp(p.datetime) - ts).total_seconds())
            if best_dist is None or d < best_dist:
                best_dist = d
                best = p
        if best is not None and best_dist <= tol * 5 * 60:  # tol bars × 5min
            return best
        return None

    @staticmethod
    def _find_matching_pivot_index(
        pivots: list, ts: pd.Timestamp, tol: int
    ) -> Optional[int]:
        """返回匹配 pivot 在列表中的索引（不是 bar 索引）。"""
        best_i = None
        best_dist = None
        for i, p in enumerate(pivots):
            d = abs((pd.Timestamp(p.datetime) - ts).total_seconds())
            if best_dist is None or d < best_dist:
                best_dist = d
                best_i = i
        if best_i is not None and best_dist <= tol * 5 * 60:
            return best_i
        return None

    @staticmethod
    def _role_to_slot(role: str) -> Optional[str]:
        """
        将结构角色映射到 ImpulseSearcher 的 anchor 键名。

        role          → slot key
        'wave1_end'   → 'w1'
        'wave2_end'   → 'w2'
        'wave3_end'   → 'w3'
        'wave4_end'   → 'w4'
        'wave5_end'   → 'w5'
        'wave0_start' → 'w0'
        """
        mapping = {
            "wave0_start": "w0",
            "wave1_end":   "w1",
            "wave2_end":   "w2",
            "wave3_end":   "w3",
            "wave4_end":   "w4",
            "wave5_end":   "w5",
        }
        return mapping.get(role)

    def clear(self):
        """清空所有锚点。"""
        self.anchors.clear()

    def __len__(self):
        return len(self.anchors)

    def __repr__(self):
        return f"AnchorManager({len(self.anchors)} anchors)"
````

````
"""
评分系统 (Scoring System)
=========================
将 Elliott Wave 的规则和指南编码为可量化的评分函数。

参考文献：
  - Frost & Prechter《Elliott Wave Principle》第2章：规则与指南
  - Neely《Mastering Elliott Wave》：更严格的规则体系
  - Bulkowski《Encyclopedia of Chart Patterns》：统计验证

规则（Rules）：违反则方案无效（硬约束）
指南（Guidelines）：违反则扣分（软约束）

PyCharm Console 调试示例：
    from ew_system.scoring import ElliottRules, score_impulse
    # 见函数文档
"""

from __future__ import annotations

import math
from typing import Dict, List, Optional, Tuple

import numpy as np


# ---------------------------------------------------------------------------
# Fibonacci 常数
# ---------------------------------------------------------------------------

FIB_RETRACE = [0.236, 0.382, 0.500, 0.618, 0.786]
FIB_EXTEND  = [1.000, 1.272, 1.618, 2.000, 2.618, 4.236]
FIB_ALL     = sorted(set(FIB_RETRACE + FIB_EXTEND))


def fib_closeness(value: float, targets: List[float], decay: float = 3.0) -> float:
    """
    计算 value 与最近 Fibonacci 目标的接近程度。

    返回 0~1 的分数：
      完全匹配 → 1.0
      距离 0.5 → exp(-3*0.5) ≈ 0.22
      距离 1.0 → exp(-3*1.0) ≈ 0.05

    参数
    ----
    value   : 待评估的比率
    targets : Fibonacci 目标列表
    decay   : 衰减系数（越大越严格）
    """
    if not targets or value <= 0:
        return 0.0
    min_dist = min(abs(value - t) for t in targets)
    return math.exp(-decay * min_dist)


# ---------------------------------------------------------------------------
# 推动浪规则检查
# ---------------------------------------------------------------------------

class ElliottRules:
    """
    Elliott Wave 推动浪（Impulse）的规则与指南。

    铁律（Rules）：违反则整个方案无效
    指南（Guidelines）：违反则扣分

    参考：Frost & Prechter《Elliott Wave Principle》
    """

    # ---- 铁律 ----

    @staticmethod
    def rule_wave2_no_exceed_wave0(
        p0: float, p1: float, p2: float, up: bool
    ) -> bool:
        """
        铁律1：浪2不能回撤超过浪1的起点（浪0）。
        上升浪：p2 > p0
        下降浪：p2 < p0
        """
        if up:
            return p2 > p0
        else:
            return p2 < p0

    @staticmethod
    def rule_wave3_not_shortest(a1: float, a3: float, a5: float) -> bool:
        """
        铁律2：浪3不能是浪1、3、5中最短的。
        （注意：不是说浪3必须最长，只是不能最短）
        """
        return not (a3 < a1 and a3 < a5)

    @staticmethod
    def rule_wave4_no_overlap_wave1(
        p1: float, p4: float, up: bool
    ) -> bool:
        """
        铁律3：浪4不能进入浪1的价格领地。
        上升浪：p4 > p1（浪4低点不能低于浪1高点）
        下降浪：p4 < p1（浪4高点不能高于浪1低点）

        例外：楔形（Diagonal Triangle）允许重叠，但这里暂不处理。
        """
        if up:
            return p4 > p1
        else:
            return p4 < p1

    @staticmethod
    def check_all_hard_rules(
        p0: float, p1: float, p2: float,
        p3: float, p4: float, p5: float,
        up: bool,
    ) -> Tuple[bool, List[str]]:
        """
        检查所有铁律。返回 (通过, 违反的规则列表)。
        """
        violations = []

        # 方向性检查
        if up:
            if not (p1 > p0): violations.append("浪1方向错误")
            if not (p2 < p1): violations.append("浪2方向错误")
            if not (p3 > p2): violations.append("浪3方向错误")
            if not (p4 < p3): violations.append("浪4方向错误")
            if not (p5 > p4): violations.append("浪5方向错误")
        else:
            if not (p1 < p0): violations.append("浪1方向错误")
            if not (p2 > p1): violations.append("浪2方向错误")
            if not (p3 < p2): violations.append("浪3方向错误")
            if not (p4 > p3): violations.append("浪4方向错误")
            if not (p5 < p4): violations.append("浪5方向错误")

        if violations:
            return False, violations

        # 计算各浪幅度
        a1 = abs(p1 - p0)
        a2 = abs(p2 - p1)
        a3 = abs(p3 - p2)
        a4 = abs(p4 - p3)
        a5 = abs(p5 - p4)

        if not ElliottRules.rule_wave2_no_exceed_wave0(p0, p1, p2, up):
            violations.append("铁律1违反：浪2超过浪0")

        if not ElliottRules.rule_wave3_not_shortest(a1, a3, a5):
            violations.append("铁律2违反：浪3是最短浪")

        if not ElliottRules.rule_wave4_no_overlap_wave1(p1, p4, up):
            violations.append("铁律3违反：浪4进入浪1领地")

        return len(violations) == 0, violations


# ---------------------------------------------------------------------------
# 推动浪评分
# ---------------------------------------------------------------------------

def score_impulse(
    p0: float, p1: float, p2: float,
    p3: float, p4: float, p5: float,
    t0: int, t1: int, t2: int,
    t3: int, t4: int, t5: int,
    up: bool,
) -> Tuple[float, Dict]:
    """
    对一个推动浪候选方案进行综合评分。

    参数
    ----
    p0~p5 : 各浪端点价格
    t0~t5 : 各浪端点的 bar 索引（用于时间比例计算）
    up    : True=上升浪, False=下降浪

    返回
    ----
    (score, details)
    score   : 0~1 的综合评分
    details : 各项指标的详细分数
    """
    details = {}
    weighted_scores = []

    # 各浪幅度
    a1 = abs(p1 - p0)
    a2 = abs(p2 - p1)
    a3 = abs(p3 - p2)
    a4 = abs(p4 - p3)
    a5 = abs(p5 - p4)

    # 各浪时间
    t_1 = t1 - t0
    t_2 = t2 - t1
    t_3 = t3 - t2
    t_4 = t4 - t3
    t_5 = t5 - t4

    # ---- G1: 浪2回撤比率（理想：0.382~0.618 of 浪1）----
    retr2 = a2 / a1 if a1 > 0 else 0
    details["retr2"] = round(retr2, 4)
    s_r2 = fib_closeness(retr2, FIB_RETRACE)
    weighted_scores.append(("retr2_fib", s_r2, 1.2))

    # ---- G2: 浪3延伸比率（理想：1.618 of 浪1）----
    ext3 = a3 / a1 if a1 > 0 else 0
    details["ext3"] = round(ext3, 4)
    s_e3 = fib_closeness(ext3, FIB_EXTEND, decay=2.5)
    weighted_scores.append(("ext3_fib", s_e3, 1.8))  # 最高权重

    # ---- G3: 浪3是最长浪（强指南）----
    if a3 >= a1 and a3 >= a5:
        s_longest = 1.0
    elif a3 >= max(a1, a5) * 0.9:
        s_longest = 0.7
    elif a3 >= a1 or a3 >= a5:
        s_longest = 0.4
    else:
        s_longest = 0.1
    details["w3_is_longest"] = s_longest
    weighted_scores.append(("w3_longest", s_longest, 1.5))

    # ---- G4: 浪4回撤比率（理想：0.236~0.382 of 浪3）----
    retr4 = a4 / a3 if a3 > 0 else 0
    details["retr4"] = round(retr4, 4)
    s_r4 = fib_closeness(retr4, [0.236, 0.382, 0.500], decay=3.5)
    weighted_scores.append(("retr4_fib", s_r4, 1.0))

    # ---- G5: 浪5/浪1比率（理想：0.618, 1.0, 1.618）----
    rat5 = a5 / a1 if a1 > 0 else 0
    details["rat5_w1"] = round(rat5, 4)
    s_r5 = fib_closeness(rat5, [0.618, 1.000, 1.618], decay=3.0)
    weighted_scores.append(("w5_w1_ratio", s_r5, 1.0))

    # ---- G6: 浪2与浪4的交替性（Alternation）----
    # 理想：一个急跌（深回撤）一个横盘（浅回撤），两者差异越大越好
    alt_diff = abs(retr2 - retr4)
    s_alt = min(1.0, alt_diff / 0.25)
    details["alternation"] = round(alt_diff, 4)
    weighted_scores.append(("alternation", s_alt, 0.8))

    # ---- G7: 时间比例（浪3时间 >= 浪1时间）----
    s_time = 0.5
    if t_1 > 0:
        if t_3 >= t_1:
            s_time += 0.3
        if 0.25 <= t_5 / t_1 <= 4.0:
            s_time += 0.2
    details["t1_bars"] = t_1
    details["t3_bars"] = t_3
    details["t5_bars"] = t_5
    weighted_scores.append(("time_prop", s_time, 0.6))

    # ---- G8: 通道线（Channel）检验 ----
    # 浪1和浪3的终点连线（基础通道），浪5应该在通道内或略超出
    # 简化版：检查浪5终点是否在浪1~浪3的价格投影范围内
    if a1 > 0 and a3 > 0:
        # 浪5目标 = 浪4终点 + 浪1幅度（等长）
        w5_target_equal = p4 + (p1 - p0) if up else p4 - (p0 - p1)
        w5_deviation = abs(p5 - w5_target_equal) / a1
        s_channel = fib_closeness(w5_deviation, [0.0, 0.382, 0.618], decay=2.0)
    else:
        s_channel = 0.5
    details["channel_score"] = round(s_channel, 4)
    weighted_scores.append(("channel", s_channel, 0.7))

    # ---- G9: 总体幅度（至少有意义的移动）----
    total_move = abs(p5 - p0)
    total_pct = total_move / abs(p0) * 100 if p0 else 0
    s_size = min(1.0, total_pct / 1.5)  # 1.5% 以上满分
    details["total_move_pct"] = round(total_pct, 3)
    weighted_scores.append(("size", s_size, 0.5))

    # ---- 综合评分 ----
    total_w = sum(w for _, _, w in weighted_scores)
    composite = sum(s * w for _, s, w in weighted_scores) / total_w if total_w else 0
    details["score_breakdown"] = {
        name: round(s, 3) for name, s, _ in weighted_scores
    }

    return round(composite, 4), details

def score_zigzag(
    pA: float, pB: float, pC: float,
    tA: int, tB: int, tC: int,
    up: bool,
) -> Tuple[float, Dict]:
    """
    对 Zigzag 调整浪（A-B-C，5-3-5 结构）进行评分。

    参数
    ----
    pA, pB, pC : A浪起点、B浪终点（A浪终点）、C浪终点
    tA, tB, tC : 对应 bar 索引
    up         : True=向上 Zigzag（B浪向上，整体向下）
                 False=向下 Zigzag（整体向上）

    注意：这里的 up 指 B 浪方向（与整体调整方向相反）
    """
    details = {}
    weighted_scores = []

    aA = abs(pB - pA)  # A浪幅度
    aB = abs(pC - pB)  # B浪幅度
    aC = abs(pC - pA) - aA + aB  # 近似C浪幅度（简化）
    aC = abs(pC - pB)  # 实际C浪幅度

    # B浪回撤比率（理想：0.382~0.786 of A浪）
    retrB = aB / aA if aA > 0 else 0
    details["retrB"] = round(retrB, 4)
    s_rB = fib_closeness(retrB, [0.382, 0.500, 0.618, 0.786])
    weighted_scores.append(("retrB_fib", s_rB, 1.2))

    # C浪与A浪比率（理想：1.0, 1.618）
    ratC = aC / aA if aA > 0 else 0
    details["ratC_A"] = round(ratC, 4)
    s_rC = fib_closeness(ratC, [0.618, 1.000, 1.618, 2.618])
    weighted_scores.append(("ratC_A", s_rC, 1.5))

    # 时间比例
    tA_dur = tB - tA
    tC_dur = tC - tB
    s_time = 0.5
    if tA_dur > 0 and 0.5 <= tC_dur / tA_dur <= 2.0:
        s_time = 0.9
    details["tA_bars"] = tA_dur
    details["tC_bars"] = tC_dur
    weighted_scores.append(("time_prop", s_time, 0.6))

    total_w = sum(w for _, _, w in weighted_scores)
    composite = sum(s * w for _, s, w in weighted_scores) / total_w if total_w else 0
    details["score_breakdown"] = {
        name: round(s, 3) for name, s, _ in weighted_scores
    }

    return round(composite, 4), details
````

````
Phase 2 — 单层浪形识别
======================
负责：
  - 在给定 pivot 集合上搜索推动浪（Impulse）候选
  - 在给定 pivot 集合上搜索调整浪（Correction）候选
  - 对每个候选方案评分（调用 scoring.py）
  - 返回 Top-K 候选，供 Phase 3 嵌套使用

算法：
  - 推动浪：DFS 枚举 6 个节点（W0~W5），剪枝 + 评分
  - 调整浪：枚举 4 个节点（A起/A终/B终/C终），评分
  - 锚点约束：structural anchor 固定某些节点

参考文献：
  - Volna et al. (2013) "Elliott Wave Theory and Genetic Algorithms"
  - Prechter & Frost《Elliott Wave Principle》第3章

class WaveSearcher:
    """
    在 pivot 集合中搜索推动浪（5浪）候选。

    算法：DFS 枚举 + 铁律剪枝 + 指南评分

    参数
    ----
    pivots    : Phase 1 输出的 Pivot 列表
    direction : 'up'（上升推动浪）或 'down'（下降推动浪）
    anchors   : 结构锚点字典，键为 'w0'~'w5'，值为 pivot 列表索引
                例：{'w3': 5, 'w4': 7} 表示第5个pivot必须是浪3终点
    top_k     : 返回最优的 K 个候选
    max_candidates : 最大枚举数量（防止组合爆炸）
    """

    def __init__(
        self,
        pivots: List[Pivot],
        direction: str = "up",
        anchors: Dict[str, Any] | None = None,
        top_k: int = 3,
        max_candidates: int = 100_000,
    ):
        self.pivots = pivots
        self.direction = direction
        self.anchors = anchors or {}
        self.top_k = top_k
        self.max_candidates = max_candidates

    def search(self) -> List[ImpulseCount]:
        """
        执行搜索，返回 Top-K 评分最高的推动浪候选。
        """
        combos = self._enumerate_candidates()
        scored = []
        for combo in combos:
            ic = self._build_and_score(combo)
            if ic is not None:
                scored.append(ic)

        scored.sort(key=lambda c: c.score, reverse=True)
        return scored[: self.top_k]

    # ------------------------------------------------------------------ #
    #  枚举候选
    # ------------------------------------------------------------------ #

    def _enumerate_candidates(self) -> List[Tuple[int, ...]]:
        """
        枚举所有满足基本约束的 6-pivot 组合。

        上升推动浪模式：L H L H L H
        下降推动浪模式：H L H L H L
        """
        n = len(self.pivots)
        if n < 6:
            return []

        pattern = (
            ["L", "H", "L", "H", "L", "H"]
            if self.direction == "up"
            else ["H", "L", "H", "L", "H", "L"]
        )

        # 为每个槽位构建候选 pivot 索引列表
        slot_candidates: List[List[int]] = []
        for slot in range(6):
            key = f"w{slot}"
            if key in self.anchors:
                # 锚点固定
                anchor_val = self.anchors[key]
                resolved = self._resolve_anchor(anchor_val)
                slot_candidates.append([resolved] if resolved is not None else [])
            else:
                slot_candidates.append([
                    i for i in range(n)
                    if self.pivots[i].kind == pattern[slot]
                ])

        # 检查是否有空槽
        for slot, cands in enumerate(slot_candidates):
            if not cands:
                return []

        # DFS 枚举
        results = []
        self._dfs(slot_candidates, 0, [], results)
        return results

    def _dfs(
        self,
        slots: List[List[int]],
        depth: int,
        current: List[int],
        results: List[Tuple],
    ):
        if len(results) >= self.max_candidates:
            return
        if depth == 6:
            results.append(tuple(current))
            return

        lo = current[-1] + 1 if current else 0
        remaining = 5 - depth

        for pi in slots[depth]:
            if pi < lo:
                continue
            # 剪枝：后面还需要 remaining 个 pivot
            if pi + remaining > len(self.pivots) - 1:
                continue
            current.append(pi)
            self._dfs(slots, depth + 1, current, results)
            current.pop()

    # ------------------------------------------------------------------ #
    #  构建并评分
    # ------------------------------------------------------------------ #

    def _build_and_score(
        self, combo: Tuple[int, ...]
    ) -> Optional[ImpulseCount]:
        """
        对一个 6-pivot 组合进行铁律检查和指南评分。
        铁律失败 → 返回 None
        铁律通过 → 返回 ImpulseCount
        """
        pvts = [self.pivots[i] for i in combo]
        prices = [p.price for p in pvts]
        times = [p.idx for p in pvts]
        p0, p1, p2, p3, p4, p5 = prices
        t0, t1, t2, t3, t4, t5 = times
        up = (self.direction == "up")

        # 铁律检查
        passed, violations = ElliottRules.check_all_hard_rules(
            p0, p1, p2, p3, p4, p5, up
        )
        if not passed:
            return None

        # 评分
        score, details = score_impulse(
            p0, p1, p2, p3, p4, p5,
            t0, t1, t2, t3, t4, t5,
            up,
        )

        return ImpulseCount(
            pivot_indices=list(combo),
            pivots=pvts,
            direction=self.direction,
            score=score,
            details=details,
        )

    def _resolve_anchor(self, anchor) -> Optional[int]:
        """将锚点值解析为 pivot 列表索引。"""
        if isinstance(anchor, int):
            return anchor if 0 <= anchor < len(self.pivots) else None
        # 当作时间戳处理
        ts = pd.Timestamp(anchor)
        best_i, best_d = None, None
        for i, p in enumerate(self.pivots):
            d = abs((p.ts - ts).total_seconds())
            if best_d is None or d < best_d:
                best_d = d
                best_i = i
        return best_i

class CorrectionSearcher:
    """
    在 pivot 集合中搜索 Zigzag 调整浪（A-B-C）候选。

    Zigzag 结构：5-3-5
      A浪：推动（5浪）
      B浪：调整（3浪）
      C浪：推动（5浪）

    在 pivot 层面简化为：4个节点（A起/A终/B终/C终）

    参数
    ----
    pivots    : Phase 1 输出的 Pivot 列表
    direction : 'down'（向下调整）或 'up'（向上调整）
    top_k     : 返回最优的 K 个候选
    """

    def __init__(
        self,
        pivots: List[Pivot],
        direction: str = "down",
        top_k: int = 3,
    ):
        self.pivots = pivots
        self.direction = direction
        self.top_k = top_k

    def search(self) -> List[CorrectionCount]:
        """搜索 Zigzag 调整浪候选。"""
        n = len(self.pivots)
        if n < 4:
            return []

        # 向下调整：H L H L（A浪向下，B浪向上，C浪向下）
        # 向上调整：L H L H
        if self.direction == "down":
            pattern = ["H", "L", "H", "L"]
        else:
            pattern = ["L", "H", "L", "H"]

        candidates = []
        for i in range(n - 3):
            for j in range(i + 1, n - 2):
                for k in range(j + 1, n - 1):
                    for m in range(k + 1, n):
                        combo = (i, j, k, m)
                        pvts = [self.pivots[x] for x in combo]
                        # 检查模式
                        if not all(
                            pvts[s].kind == pattern[s] for s in range(4)
                        ):
                            continue
                        cc = self._score_zigzag(combo, pvts)
                        if cc is not None:
                            candidates.append(cc)

        candidates.sort(key=lambda c: c.score, reverse=True)
        return candidates[: self.top_k]

    def _score_zigzag(
        self,
        combo: Tuple[int, ...],
        pvts: List[Pivot],
    ) -> Optional[CorrectionCount]:
        """对 Zigzag 候选评分。"""
        pA, pB, pC, pD = [p.price for p in pvts]
        tA, tB, tC, tD = [p.idx for p in pvts]
        up_b = (self.direction == "down")  # B浪方向

        # 基本方向检查
        if self.direction == "down":
            if not (pB < pA and pC > pB and pD < pC):
                return None
        else:
            if not (pB > pA and pC < pB and pD > pC):
                return None

        score, details = score_zigzag(pA, pB, pD, tA, tB, tD, up_b)

        return CorrectionCount(
            pivot_indices=list(combo),
            pivots=pvts,
            correction_type="zigzag",
            direction=self.direction,
            score=score,
            details=details,
        )
````

````
"""
Phase 3 — 嵌套验证与约束传递
==============================
这是整个系统最核心的部分。

核心思想（自顶向下嵌套，Top-Down Decomposition）：
  1. 在最大尺度（如日线/4H）识别大级别浪形
  2. 对大级别浪形的每个子浪，切出对应时间段的更细粒度数据
  3. 在细粒度数据上再次运行 Phase 1-2，但受父浪约束
  4. 约束传递：
     - 子浪的起终点必须对齐父浪的 pivot
     - 子浪的方向由父浪决定
     - 子浪的价格范围受父浪限制
     - 子浪的时间范围受父浪限制
  5. 递归到所需深度

参考文献：
  - Frost & Prechter《Elliott Wave Principle》第4章：嵌套结构
  - Neely《Mastering Elliott Wave》：严格的子浪约束规则
  - 多时间框架分析（MTF）：大级别约束小级别
  - Poser (2003) "Applying Elliott Wave Theory Profitably"

关键约束（来自 Neely NeoWave）：
  1. 子浪1的起点 = 父浪起点（价格和时间完全对齐）
  2. 子浪5的终点 = 父浪终点（价格和时间完全对齐）
  3. 子浪的总时间 = 父浪的总时间
  4. 推动浪内部：奇数浪（1,3,5）为推动，偶数浪（2,4）为调整
  5. 调整浪内部：A和C为推动，B为调整

PyCharm Console 调试示例：

    result = analyzer.analyze(max_depth=2)

@dataclass
class WaveNode:
    """
    波浪树的一个节点，代表某一级别的一段浪形。

    属性
    ----
    level       : 层级深度（0=最大级别，1=子浪，2=子子浪...）
    wave_label  : 浪号标签（如 '1', '2', '3', '4', '5', 'A', 'B', 'C'）
    wave_type   : 'impulse' 或 'correction'
    direction   : 'up' 或 'down'
    start_pivot : 起点 Pivot
    end_pivot   : 终点 Pivot
    score       : 该节点的评分
    children    : 子浪节点列表（下一层级）
    parent      : 父节点引用
    tf          : 所属时间框架（如 '1h', '4h', 'D'）
    count       : 对应的 ImpulseCount 或 CorrectionCount 对象
    """
    level: int
    wave_label: str
    wave_type: str                  # 'impulse' or 'correction'
    direction: str                  # 'up' or 'down'
    start_pivot: Pivot
    end_pivot: Pivot
    score: float = 0.0
    children: List["WaveNode"] = field(default_factory=list)
    parent: Optional["WaveNode"] = None
    tf: str = ""
    count: Any = None               # ImpulseCount or CorrectionCount

    @property
    def start_time(self) -> pd.Timestamp:
        return self.start_pivot.ts

    @property
    def end_time(self) -> pd.Timestamp:
        return self.end_pivot.ts

    @property
    def start_price(self) -> float:
        return self.start_pivot.price

    @property
    def end_price(self) -> float:
        return self.end_pivot.price

    @property
    def price_move(self) -> float:
        return self.end_price - self.start_price

    @property
    def duration_bars(self) -> int:
        return self.end_pivot.idx - self.start_pivot.idx

    def add_child(self, child: "WaveNode"):
        child.parent = self
        self.children.append(child)

    def __repr__(self):
        d = "↑" if self.direction == "up" else "↓"
        return (f"WaveNode(L{self.level} {self.wave_label}{d} "
                f"score={self.score:.3f} "
                f"{self.start_time.strftime('%m-%d %H:%M')}"
                f"→{self.end_time.strftime('%m-%d %H:%M')} "
                f"{self.start_price:.2f}→{self.end_price:.2f})")


@dataclass
class NestedResult:
    """
    Phase 3 的完整分析结果。

    包含：
      - 多个顶层候选（Top-K 大级别浪形）
      - 每个候选的完整波浪树
      - 综合置信度评分
      - 当前位置判断
    """
    top_candidates: List[WaveNode] = field(default_factory=list)
    current_position: Optional[str] = None   # 如 "主浪③中的小浪④"
    confidence: float = 0.0
    timeframes_used: List[str] = field(default_factory=list)
    analysis_log: List[str] = field(default_factory=list)

    def print_tree(self, max_depth: int = 99):
        """打印波浪树结构。"""
        print("=" * 70)
        print("Phase 3 嵌套分析结果")
        print("=" * 70)
        print(f"置信度: {self.confidence:.3f}")
        print(f"当前位置: {self.current_position or '未确定'}")
        print(f"使用时间框架: {', '.join(self.timeframes_used)}")
        print()

        for i, root in enumerate(self.top_candidates):
            print(f"--- 候选 #{i+1} (score={root.score:.4f}) ---")
            self._print_node(root, indent=0, max_depth=max_depth)
            print()

    def _print_node(self, node: WaveNode, indent: int, max_depth: int):
        if node.level > max_depth:
            return
        prefix = "  " * indent
        d = "↑" if node.direction == "up" else "↓"
        print(f"{prefix}[{node.wave_label}]{d} "
              f"{node.start_time.strftime('%m-%d %H:%M')}"
              f"→{node.end_time.strftime('%m-%d %H:%M')} "
              f"{node.start_price:.4f}→{node.end_price:.4f} "
              f"score={node.score:.3f} tf={node.tf}")
        for child in node.children:
            self._print_node(child, indent + 1, max_depth)

    def get_current_wave_path(self) -> List[WaveNode]:
        """
        返回当前价格所在的波浪路径（从根到叶）。
        用于确定"当前处于哪个级别的哪个浪"。
        """
        if not self.top_candidates:
            return []
        root = self.top_candidates[0]
        return self._find_current_path(root)

    def _find_current_path(self, node: WaveNode) -> List[WaveNode]:
        """找到最后一个子节点的路径（代表当前位置）。"""
        if not node.children:
            return [node]
        last_child = node.children[-1]
        return [node] + self._find_current_path(last_child)

    # 时间约束
    start_time: pd.Timestamp
    end_time: pd.Timestamp
    start_bar_idx: int              # 在源 DataFrame 中的 bar 索引
    end_bar_idx: int

    # 价格约束

    end_price: float

    parent_wave_label: str          # 父浪标签

    parent_atr_mult: float = 2.0

    def time_slice(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        从源 DataFrame 中切出该约束对应的时间段。
        使用 bar 索引确保精确对齐。

        mask = (
            (df.index >= self.start_bar_idx) &
            (df.index <= self.end_bar_idx)
        )
        sliced = df[mask].copy().reset_index(drop=True)
        return sliced

    def is_valid(self) -> bool:
        """检查约束是否有效（时间范围至少有足够的 bar）。"""
        return (self.end_bar_idx - self.start_bar_idx) >= 6


# ---------------------------------------------------------------------------
# 核心嵌套分析器
# ---------------------------------------------------------------------------

class NestedAnalyzer:
    """
    Phase 3 核心：自顶向下嵌套波浪分析。

    工作流程
    --------
    1. 在最粗粒度（大 atr_mult）识别顶层浪形
    2. 对顶层浪形的每个子浪段，切出对应时间段
    3. 在子时间段上用更小的 atr_mult 重新识别 pivot
    4. 在子 pivot 上搜索符合约束的子浪形
    5. 递归到 max_depth 层

    参数
    ----
    df          : 标准化的 5-min OHLCV DataFrame（Phase 0 输出）
    atr_scales  : 各层级使用的 ATR 倍数列表（从大到小）
                  例：[5.0, 2.5, 1.2] 表示3层嵌套
    top_k       : 每层保留的最优候选数
    max_depth   : 最大嵌套深度
    direction   : 顶层方向（'up', 'down', 'auto'）
    """

    # 各层级的标签系统
    _IMPULSE_LABELS = {
        0: ["Ⅰ", "Ⅱ", "Ⅲ", "Ⅳ", "Ⅴ"],          # 超级周期
        1: ["①", "②", "③", "④", "⑤"],             # 周期级
        2: ["1", "2", "3", "4", "5"],               # 主浪级
        3: ["i", "ii", "iii", "iv", "v"],           # 中级
        4: ["[i]", "[ii]", "[iii]", "[iv]", "[v]"], # 小级
    }
    _CORRECTION_LABELS = {
        0: ["A", "B", "C"],
        1: ["a", "b", "c"],
        2: ["(a)", "(b)", "(c)"],
        3: ["[a]", "[b]", "[c]"],
    }

    def __init__(
        self,
        df: pd.DataFrame,
        atr_scales: List[float] | None = None,
        top_k: int = 3,
        max_depth: int = 3,
        direction: str = "auto",
        atr_period: int = 14,
    ):
        self.df = df.copy().reset_index(drop=True)
        self.atr_period = atr_period
        self.top_k = top_k
        self.max_depth = max_depth
        self.direction = direction

        # 默认尺度：从大到小，每层约缩小2倍
        if atr_scales is None:
            atr_scales = [5.0, 2.5, 1.2, 0.8]
        self.atr_scales = atr_scales[:max_depth + 1]

        self._log: List[str] = []

    def analyze(self) -> NestedResult:
        """
        执行完整的嵌套分析。

        返回 NestedResult，包含波浪树和当前位置判断。
        """
        self._log = []
        self._log.append(f"开始 Phase 3 嵌套分析，最大深度={self.max_depth}")
        self._log.append(f"ATR 尺度序列: {self.atr_scales}")

        # ---- Step 1: 顶层分析 ----
        top_scale = self.atr_scales[0]
        self._log.append(f"\n[Level 0] 顶层分析 (ATR×{top_scale})")

        top_pivots, top_atr = self._get_pivots(self.df, top_scale)
        self._log.append(f"  顶层 pivots: {len(top_pivots)} 个")

        if len(top_pivots) < 6:
            self._log.append("  ⚠ pivot 数量不足，尝试减小 atr_scales[0]")
            return NestedResult(
                analysis_log=self._log,
                confidence=0.0,
            )

        # 自动检测方向
        direction = self.direction
        if direction == "auto":
            direction = self._detect_direction(top_pivots)
        self._log.append(f"  检测到方向: {direction}")

        # 搜索顶层推动浪
        top_searcher = WaveSearcher(
            top_pivots, direction=direction, top_k=self.top_k
        )
        top_candidates = top_searcher.search()

        if not top_candidates:
            # 尝试反方向
            alt_dir = "down" if direction == "up" else "up"
            self._log.append(f"  {direction} 方向无结果，尝试 {alt_dir}")
            top_searcher2 = WaveSearcher(
                top_pivots, direction=alt_dir, top_k=self.top_k
            )
            top_candidates = top_searcher2.search()
            if top_candidates:
                direction = alt_dir

        if not top_candidates:
            self._log.append("  ⚠ 顶层未找到有效推动浪")
            return NestedResult(analysis_log=self._log, confidence=0.0)

        self._log.append(f"  找到 {len(top_candidates)} 个顶层候选")

        # ---- Step 2: 递归嵌套 ----
        result_nodes = []
        for rank, top_count in enumerate(top_candidates):
            self._log.append(
                f"\n[Level 0] 候选 #{rank+1} score={top_count.score:.4f}"
            )
            root_node = self._build_root_node(top_count, direction, level=0)
            if self.max_depth > 0:
                self._decompose_impulse(root_node, top_count, depth=1)
            result_nodes.append(root_node)

        # ---- Step 3: 综合评分与当前位置判断 ----
        confidence = self._compute_confidence(result_nodes)
        current_pos = self._determine_current_position(result_nodes)

        result = NestedResult(
            top_candidates=result_nodes,
            current_position=current_pos,
            confidence=confidence,
            timeframes_used=[f"ATR×{s}" for s in self.atr_scales],
            analysis_log=self._log,
        )

        self._log.append(f"\n分析完成，置信度={confidence:.3f}")
        self._log.append(f"当前位置: {current_pos}")

        return result

    # ------------------------------------------------------------------ #
    #  递归分解
    # ------------------------------------------------------------------ #

    def _decompose_impulse(
        self,
        parent_node: WaveNode,
        parent_count: ImpulseCount,
        depth: int,
    ):
        """
        将一个推动浪分解为5个子浪，并递归分解每个子浪。

        子浪约束：
          - 奇数子浪（1,3,5）：推动浪，方向与父浪相同
          - 偶数子浪（2,4）：调整浪，方向与父浪相反
          - 子浪1起点 = 父浪起点（W0）
          - 子浪5终点 = 父浪终点（W5）
          - 相邻子浪的端点必须对齐
        """
        if depth > self.max_depth:
            return
        if depth >= len(self.atr_scales):
            return

        child_scale = self.atr_scales[depth]
        parent_dir = parent_count.direction
        child_dir_motive = parent_dir
        child_dir_corrective = "down" if parent_dir == "up" else "up"

        # 5个子浪的端点对
        wave_endpoints = [
            (parent_count.w0, parent_count.w1, "1", "impulse",    child_dir_motive),
            (parent_count.w1, parent_count.w2, "2", "correction", child_dir_corrective),
            (parent_count.w2, parent_count.w3, "3", "impulse",    child_dir_motive),
            (parent_count.w3, parent_count.w4, "4", "correction", child_dir_corrective),
            (parent_count.w4, parent_count.w5, "5", "impulse",    child_dir_motive),
        ]

        label_set = self._IMPULSE_LABELS.get(depth, ["1", "2", "3", "4", "5"])

        for seg_idx, (p_start, p_end, raw_label, wtype, wdir) in enumerate(wave_endpoints):
            label = label_set[seg_idx] if seg_idx < len(label_set) else raw_label

            # 切出子时间段
            constraints = self._build_constraints(
                p_start, p_end, wdir, wtype, child_scale
            )
            if not constraints.is_valid():
                self._log.append(
                    f"  [L{depth}] 子浪{label} 时间段太短，跳过"
                )
                # 仍然创建节点，但无子浪
                child_node = WaveNode(
                    level=depth,
                    wave_label=label,
                    wave_type=wtype,
                    direction=wdir,
                    start_pivot=p_start,
                    end_pivot=p_end,
                    score=0.5,
                    tf=f"ATR×{child_scale}",
                )
                parent_node.add_child(child_node)
                continue

            sub_df = constraints.time_slice(self.df)
            if len(sub_df) < 6:
                child_node = WaveNode(
                    level=depth,
                    wave_label=label,
                    wave_type=wtype,
                    direction=wdir,
                    start_pivot=p_start,
                    end_pivot=p_end,
                    score=0.5,
                    tf=f"ATR×{child_scale}",
                )
                parent_node.add_child(child_node)
                continue

            # 在子时间段上识别 pivot
            sub_pivots, _ = self._get_pivots(sub_df, child_scale)

            # 强制对齐端点
            sub_pivots = self._align_endpoints(
                sub_pivots, sub_df, p_start, p_end, wdir
            )

            self._log.append(
                f"  [L{depth}] 子浪{label}({wtype},{wdir}): "
                f"{len(sub_df)} bars, {len(sub_pivots)} pivots"
            )

            if wtype == "impulse" and len(sub_pivots) >= 6:
                # 搜索子推动浪
                sub_searcher = WaveSearcher(
                    sub_pivots, direction=wdir, top_k=1
                )
                sub_cands = sub_searcher.search()

                if sub_cands:
                    best = sub_cands[0]
                    child_node = WaveNode(
                        level=depth,
                        wave_label=label,
                        wave_type=wtype,
                        direction=wdir,
                        start_pivot=p_start,
                        end_pivot=p_end,
                        score=best.score,
                        tf=f"ATR×{child_scale}",
                        count=best,
                    )
                    parent_node.add_child(child_node)

                    # 递归分解
                    if depth < self.max_depth:
                        self._decompose_impulse(child_node, best, depth + 1)
                else:
                    # 未找到子推动浪，创建占位节点
                    child_node = WaveNode(
                        level=depth,
                        wave_label=label,
                        wave_type=wtype,
                        direction=wdir,
                        start_pivot=p_start,
                        end_pivot=p_end,
                        score=0.3,
                        tf=f"ATR×{child_scale}",
                    )
                    parent_node.add_child(child_node)

            elif wtype == "correction" and len(sub_pivots) >= 4:
                # 搜索子调整浪（Zigzag）
                corr_dir = wdir
                sub_corr = CorrectionSearcher(
                    sub_pivots, direction=corr_dir, top_k=1
                )
                corr_cands = sub_corr.search()

                if corr_cands:
                    best_corr = corr_cands[0]
                    child_node = WaveNode(
                        level=depth,
                        wave_label=label,
                        wave_type=wtype,
                        direction=wdir,
                        start_pivot=p_start,
                        end_pivot=p_end,
                        score=best_corr.score,
                        tf=f"ATR×{child_scale}",
                        count=best_corr,
                    )
                    parent_node.add_child(child_node)
                else:
                    child_node = WaveNode(
                        level=depth,
                        wave_label=label,
                        wave_type=wtype,
                        direction=wdir,
                        start_pivot=p_start,
                        end_pivot=p_end,
                        score=0.3,
                        tf=f"ATR×{child_scale}",
                    )
                    parent_node.add_child(child_node)
            else:
                # pivot 不足，创建占位节点
                child_node = WaveNode(
                    level=depth,
                    wave_label=label,
                    wave_type=wtype,
                    direction=wdir,
                    start_pivot=p_start,
                    end_pivot=p_end,
                    score=0.2,
                    tf=f"ATR×{child_scale}",
                )
                parent_node.add_child(child_node)

    #  辅助方法

    def _get_pivots(
        self, df: pd.DataFrame, atr_mult: float
    ) -> Tuple[List[Pivot], pd.Series]:
        """在给定 DataFrame 上用指定 ATR 倍数识别 pivot。"""
        engine = PivotEngine(
            atr_period=min(self.atr_period, max(3, len(df) // 5)),
            atr_mult=atr_mult,
        )
        return engine.find_pivots(df)

    def _build_constraints(
        self,
        p_start: Pivot,
        p_end: Pivot,
        direction: str,
        wave_type: str,
        child_scale: float,
    ) -> WaveConstraints:
        """从父浪的两个端点构建子浪约束。"""
        return WaveConstraints(
            start_time=p_start.ts,
            end_time=p_end.ts,
            start_bar_idx=p_start.idx,
            end_bar_idx=p_end.idx,
            start_price=p_start.price,
            end_price=p_end.price,
            direction=direction,
            wave_type=wave_type,
            parent_wave_label="",
            suggested_child_atr_mult=child_scale,
        )

    def _align_endpoints(
        self,
        pivots: List[Pivot],
        sub_df: pd.DataFrame,
        p_start: Pivot,
        p_end: Pivot,
        direction: str,
    ) -> List[Pivot]:
        """
        强制对齐子浪的起终点与父浪端点。

        Elliott Wave 嵌套的核心约束：
          子浪1的起点 = 父浪起点
          子浪5的终点 = 父浪终点

        实现方式：
          1. 如果 pivot 列表的第一个点不是父浪起点，插入父浪起点
          2. 如果 pivot 列表的最后一个点不是父浪终点，插入父浪终点
          3. 重新强制交替
        """
        if not pivots:
            return pivots

        # 确定起点和终点的 kind
        if direction == "up":
            start_kind = "L"
            end_kind = "H"
        else:
            start_kind = "H"
            end_kind = "L"

        # 创建对齐的起点 pivot（使用子 df 中的相对索引）
        # 注意：sub_df 已经 reset_index，所以 idx 从 0 开始
        start_aligned = Pivot(
            idx=0,
            datetime=p_start.datetime,
            price=p_start.price,
            kind=start_kind,
            atr=p_start.atr,
        )
        end_aligned = Pivot(
            idx=len(sub_df) - 1,
            datetime=p_end.datetime,
            price=p_end.price,
            kind=end_kind,
            atr=p_end.atr,
        )

        # 检查并替换/插入起点
        if pivots[0].idx > 1:
            pivots.insert(0, start_aligned)
        else:
            # 替换第一个 pivot 为对齐版本
            pivots[0] = start_aligned

        # 检查并替换/插入终点
        if pivots[-1].idx < len(sub_df) - 2:
            pivots.append(end_aligned)
        else:
            pivots[-1] = end_aligned

        # 重新强制交替
        return PivotEngine._enforce_alternation(pivots)

    def _build_root_node(
        self,
        count: ImpulseCount,
        direction: str,
        level: int,
    ) -> WaveNode:
        """从顶层 ImpulseCount 构建根节点。"""
        label_set = self._IMPULSE_LABELS.get(level, ["1", "2", "3", "4", "5"])
        return WaveNode(
            level=level,
            wave_label=f"完整推动浪({label_set[0]}~{label_set[4]})",
            wave_type="impulse",
            direction=direction,
            start_pivot=count.w0,
            end_pivot=count.w5,
            score=count.score,
            tf=f"ATR×{self.atr_scales[0]}",
            count=count,
        )

    def _detect_direction(self, pivots: List[Pivot]) -> str:
        """根据 pivot 序列的整体趋势自动检测方向。"""
        if len(pivots) < 2:
            return "up"
        first_price = pivots[0].price
        last_price = pivots[-1].price
        return "up" if last_price > first_price else "down"

    def _compute_confidence(self, nodes: List[WaveNode]) -> float:
        """
        计算整体置信度。

        综合考虑：
          1. 顶层候选的评分
          2. 子浪嵌套的完整性
          3. 各层级评分的一致性
        """
        if not nodes:
            return 0.0

        # 顶层最优候选的评分
        top_score = nodes[0].score if nodes else 0.0

        # 子浪完整性（有多少子浪成功识别）
        total_children = 0
        scored_children = 0
        for node in nodes[:1]:  # 只看最优候选
            for child in node.children:
                total_children += 1
                if child.score > 0.4:
                    scored_children += 1

        completeness = scored_children / total_children if total_children > 0 else 0.5

        # 综合
        confidence = top_score * 0.6 + completeness * 0.4
        return round(confidence, 4)

    def _determine_current_position(
        self, nodes: List[WaveNode]
    ) -> str:
        """
        判断当前价格所处的波浪位置。

        逻辑：找到最后一个子节点（代表最新的价格位置），
        然后沿路径向上描述位置。
        """
        if not nodes:
            return "未确定"

        root = nodes[0]
        path = self._get_leaf_path(root)

        if not path:
            return "未确定"

        # 构建描述字符串
        parts = []
        for node in path:
            d = "↑" if node.direction == "up" else "↓"
            parts.append(f"{node.wave_label}{d}")

        return " → ".join(parts)

    def _get_leaf_path(self, node: WaveNode) -> List[WaveNode]:
        """获取从根到最后一个叶节点的路径。"""
        if not node.children:
            return [node]
        last_child = node.children[-1]
        return [node] + self._get_leaf_path(last_child)

class MultiTFNestedAnalyzer:
    """
    基于多时间框架的嵌套分析器。

    与 NestedAnalyzer 的区别：
      - NestedAnalyzer：在同一 5min 数据上用不同 ATR 倍数模拟多尺度
      - MultiTFNestedAnalyzer：真正使用不同频率的 OHLCV 数据
        （日线 → 4H → 1H → 30min → 5min）

    这更接近实际交易中的多时间框架分析（MTF）。

    参数
    ----
    multi_tf_data : dict
        {'D': df_daily, '4h': df_4h, '1h': df_1h, '5min': df_5min}
    top_k         : 每层保留的最优候选数
    max_depth     : 最大嵌套深度
    """

    # 时间框架的层级顺序（从粗到细）
    TF_HIERARCHY = ["M", "W", "D", "4h", "1h", "30min", "15min", "5min"]

    # 每个时间框架对应的 ATR 倍数
    TF_ATR_MULT = {
        "M": 3.0, "W": 2.5, "D": 2.0,
        "4h": 1.8, "1h": 1.5, "30min": 1.2, "15min": 1.0, "5min": 0.8,
    }

    def __init__(
        self,
        multi_tf_data: Dict[str, pd.DataFrame],
        top_k: int = 3,
        max_depth: int = 3,
        direction: str = "auto",
    ):
        self.data = multi_tf_data
        self.top_k = top_k
        self.max_depth = max_depth
        self.direction = direction

        # 确定可用的时间框架（按层级顺序）
        self.available_tfs = [
            tf for tf in self.TF_HIERARCHY
            if tf in multi_tf_data and len(multi_tf_data[tf]) >= 6
        ]
        self._log: List[str] = []

    def analyze(self) -> NestedResult:
        """执行多时间框架嵌套分析。"""
        self._log = []

        if not self.available_tfs:
            return NestedResult(
                analysis_log=["⚠ 没有可用的时间框架数据"],
                confidence=0.0,
            )

        self._log.append(f"可用时间框架: {self.available_tfs}")

        # 从最粗粒度开始
        top_tf = self.available_tfs[0]
        top_df = self.data[top_tf]
        top_atr_mult = self.TF_ATR_MULT.get(top_tf, 2.0)

        self._log.append(f"\n[Level 0] 顶层分析 TF={top_tf}, ATR×{top_atr_mult}")

        # 识别顶层 pivot
        engine = PivotEngine(atr_period=14, atr_mult=top_atr_mult)
        top_pivots, _ = engine.find_pivots(top_df)
        self._log.append(f"  顶层 pivots: {len(top_pivots)} 个")

        if len(top_pivots) < 6:
            return NestedResult(
                analysis_log=self._log,
                confidence=0.0,
            )

        # 检测方向
        direction = self.direction
        if direction == "auto":
            direction = "up" if top_pivots[-1].price > top_pivots[0].price else "down"

        # 搜索顶层推动浪
        searcher = WaveSearcher(top_pivots, direction=direction, top_k=self.top_k)
        top_candidates = searcher.search()

        if not top_candidates:
            return NestedResult(analysis_log=self._log, confidence=0.0)

        # 构建结果树
        result_nodes = []
        for rank, count in enumerate(top_candidates):
            root = WaveNode(
                level=0,
                wave_label=f"顶层推动浪",
                wave_type="impulse",
                direction=direction,
                start_pivot=count.w0,
                end_pivot=count.w5,
                score=count.score,
                tf=top_tf,
                count=count,
            )

            # 递归分解（使用下一个时间框架）
            if len(self.available_tfs) > 1 and self.max_depth > 0:
                self._decompose_with_finer_tf(root, count, depth=1)

            result_nodes.append(root)

        confidence = self._compute_confidence(result_nodes)
        current_pos = self._determine_current_position(result_nodes)

        return NestedResult(
            top_candidates=result_nodes,
            current_position=current_pos,
            confidence=confidence,
            timeframes_used=self.available_tfs,
            analysis_log=self._log,
        )

    def _decompose_with_finer_tf(
        self,
        parent_node: WaveNode,
        parent_count: ImpulseCount,
        depth: int,
    ):
        """使用更细粒度的时间框架分解子浪。"""
        if depth > self.max_depth or depth >= len(self.available_tfs):
            return

        child_tf = self.available_tfs[depth]
        child_df = self.data[child_tf]
        child_atr_mult = self.TF_ATR_MULT.get(child_tf, 1.5)

        parent_dir = parent_count.direction
        child_dir_motive = parent_dir
        child_dir_corrective = "down" if parent_dir == "up" else "up"

        wave_endpoints = [
            (parent_count.w0, parent_count.w1, "1", "impulse",    child_dir_motive),
            (parent_count.w1, parent_count.w2, "2", "correction", child_dir_corrective),
            (parent_count.w2, parent_count.w3, "3", "impulse",    child_dir_motive),
            (parent_count.w3, parent_count.w4, "4", "correction", child_dir_corrective),
            (parent_count.w4, parent_count.w5, "5", "impulse",    child_dir_motive),
        ]

        label_map = {0: "1", 1: "2", 2: "3", 3: "4", 4: "5"}

        for seg_idx, (p_start, p_end, raw_label, wtype, wdir) in enumerate(wave_endpoints):
            label = label_map[seg_idx]

            # 在细粒度数据中切出时间段
            mask = (
                (child_df["datetime"] >= p_start.ts) &
                (child_df["datetime"] <= p_end.ts)
            )
            sub_df = child_df[mask].copy().reset_index(drop=True)

            self._log.append(
                f"  [L{depth}] 子浪{label}({wtype},{wdir}) "
                f"TF={child_tf}: {len(sub_df)} bars"
            )

            if len(sub_df) < 6:
                child_node = WaveNode(
                    level=depth, wave_label=label, wave_type=wtype,
                    direction=wdir, start_pivot=p_start, end_pivot=p_end,
                    score=0.3, tf=child_tf,
                )
                parent_node.add_child(child_node)
                continue

            # 识别子 pivot
            engine = PivotEngine(
                atr_period=min(14, max(3, len(sub_df) // 4)),
                atr_mult=child_atr_mult,
            )
            sub_pivots, _ = engine.find_pivots(sub_df)

            if wtype == "impulse" and len(sub_pivots) >= 6:
                sub_searcher = WaveSearcher(sub_pivots, direction=wdir, top_k=1)
                sub_cands = sub_searcher.search()
                if sub_cands:
                    best = sub_cands[0]
                    child_node = WaveNode(
                        level=depth, wave_label=label, wave_type=wtype,
                        direction=wdir, start_pivot=p_start, end_pivot=p_end,
                        score=best.score, tf=child_tf, count=best,
                    )
                    parent_node.add_child(child_node)
                    if depth < self.max_depth:
                        self._decompose_with_finer_tf(child_node, best, depth + 1)
                    continue

            elif wtype == "correction" and len(sub_pivots) >= 4:
                corr_searcher = CorrectionSearcher(sub_pivots, direction=wdir, top_k=1)
                corr_cands = corr_searcher.search()
                if corr_cands:
                    best_corr = corr_cands[0]
                    child_node = WaveNode(
                        level=depth, wave_label=label, wave_type=wtype,
                        direction=wdir, start_pivot=p_start, end_pivot=p_end,
                        score=best_corr.score, tf=child_tf, count=best_corr,
                    )
                    parent_node.add_child(child_node)
                    continue

            # 占位节点
            child_node = WaveNode(
                level=depth, wave_label=label, wave_type=wtype,
                direction=wdir, start_pivot=p_start, end_pivot=p_end,
                score=0.3, tf=child_tf,
            )
            parent_node.add_child(child_node)

    def _compute_confidence(self, nodes: List[WaveNode]) -> float:
        if not nodes:
            return 0.0
        top_score = nodes[0].score
        total_children = sum(len(n.children) for n in nodes[:1])
        scored_children = sum(
            1 for n in nodes[:1]
            for c in n.children if c.score > 0.4
        )
        completeness = scored_children / total_children if total_children > 0 else 0.5
        return round(top_score * 0.6 + completeness * 0.4, 4)

    def _determine_current_position(self, nodes: List[WaveNode]) -> str:
        if not nodes:
            return "未确定"
        root = nodes[0]
        path = []
        node = root
        while node.children:
            node = node.children[-1]
            path.append(node)
        if not path:
            return f"顶层推动浪 ({root.tf})"
        parts = [f"{n.wave_label}({'↑' if n.direction == 'up' else '↓'})" for n in path]
        return " → ".join(parts)
````

````
    def _draw_wave_node(
        self,
        ax,
        node,
        depth: int,
        max_depth: int,
    ):
        """递归绘制波浪节点。"""
        if depth > max_depth:
            return

        color = WAVE_COLORS[depth % len(WAVE_COLORS)]
        lw = max(0.5, 2.5 - depth * 0.5)
        alpha = max(0.4, 0.9 - depth * 0.15)

        # 绘制该节点的连线（起点→终点）
        ax.plot(
            [node.start_time, node.end_time],
            [node.start_price, node.end_price],
            color=color, linewidth=lw, alpha=alpha, zorder=3 + depth
        )

        # 标注浪号
        mid_time = node.start_time + (node.end_time - node.start_time) / 2
        mid_price = (node.start_price + node.end_price) / 2
        ax.annotate(
            node.wave_label,
            (mid_time, mid_price),
            fontsize=max(6, 10 - depth * 1.5),
            color=color,
            ha="center", va="center",
            bbox=dict(boxstyle="round,pad=0.15", fc="white", ec=color, alpha=0.7),
            zorder=8 + depth,
        )

        # 递归绘制子节点
        for child in node.children:
            self._draw_wave_node(ax, child, depth + 1, max_depth)
````

````
"""
端到端 Pipeline
===============
将 Phase 0-3 串联起来的高层接口。

支持两种使用模式：
  1. 快速模式：一行代码完成全部分析
  2. 分步模式：在 PyCharm Console 中逐步调试

PyCharm Console 快速使用：
    from ew_system.pipeline import Pipeline
    pipe = Pipeline("futu_data/K_5M/US_QQQ_K_5M_2026-04-22_2026-04-24.csv")
    pipe.run()
    pipe.print_report()
    pipe.plot()

分步调试：
    from ew_system.pipeline import Pipeline
    pipe = Pipeline("data.csv", atr_mult=1.5, max_depth=2)
    pipe.run_phase0()   # 只运行数据加载
    pipe.run_phase1()   # 只运行 pivot 识别
    pipe.run_phase2()   # 只运行浪形搜索
    pipe.run_phase3()   # 只运行嵌套分析
"""

from __future__ import annotations

from pathlib import Path
from typing import Any, Dict, List, Optional

import pandas as pd

from ew_system.phase0_data import DataLoader
from ew_system.phase1_pivots import PivotEngine, MultiScalePivots, Pivot
from ew_system.phase2_waves import (
    ImpulseCount, CorrectionCount,
    WaveSearcher, CorrectionSearcher
)
from ew_system.phase3_nested import (
    NestedAnalyzer, MultiTFNestedAnalyzer, NestedResult
)
from ew_system.anchors import AnchorManager
from ew_system.visualize import WavePlotter


class Pipeline:
    """
    Elliott Wave 分析端到端 Pipeline。

    参数
    ----
    csv_path    : CSV 文件路径
    atr_period  : ATR 计算周期（默认 14）
    atr_mult    : Phase 1 的 ATR 倍数（默认 2.0）
    direction   : 顶层方向（'up', 'down', 'auto'）
    top_k       : 每层保留的最优候选数（默认 3）
    max_depth   : Phase 3 嵌套深度（默认 2）
    atr_scales  : Phase 3 各层 ATR 倍数列表
    anchor_mgr  : 锚点管理器（可选）
    use_multi_tf: 是否使用多时间框架嵌套（默认 False，使用 ATR 多尺度）
    """

    def __init__(
        self,
        csv_path: str | Path,
        atr_period: int = 14,
        atr_mult: float = 2.0,
        direction: str = "auto",
        top_k: int = 3,
        max_depth: int = 2,
        atr_scales: List[float] | None = None,
        anchor_mgr: AnchorManager | None = None,
        use_multi_tf: bool = False,
    ):
        self.csv_path = Path(csv_path)
        self.atr_period = atr_period
        self.atr_mult = atr_mult
        self.direction = direction
        self.top_k = top_k
        self.max_depth = max_depth
        self.atr_scales = atr_scales or [4.0, 2.0, 1.0]
        self.anchor_mgr = anchor_mgr or AnchorManager()
        self.use_multi_tf = use_multi_tf

        # 结果存储
        self.loader: Optional[DataLoader] = None
        self.df: pd.DataFrame = pd.DataFrame()
        self.multi_tf: Dict[str, pd.DataFrame] = {}
        self.pivots: List[Pivot] = []
        self.atr: pd.Series = pd.Series(dtype=float)
        self.candidates_up: List[ImpulseCount] = []
        self.candidates_dn: List[ImpulseCount] = []
        self.candidates: List[ImpulseCount] = []
        self.corrections: List[CorrectionCount] = []
        self.nested_result: Optional[NestedResult] = None
        self.msp: Optional[MultiScalePivots] = None

    # ------------------------------------------------------------------ #
    #  分步运行（PyCharm Console 调试友好）
    # ------------------------------------------------------------------ #

    def run_phase0(self) -> pd.DataFrame:
        """Phase 0：加载并验证数据。"""
        print("[Phase 0] 加载数据...")
        self.loader = DataLoader(self.csv_path)
        self.df = self.loader.load()

        print(f"  ✔ {len(self.df)} 根K线")
        print(f"  时间范围: {self.df['datetime'].iloc[0]} → "
              f"{self.df['datetime'].iloc[-1]}")
        print(f"  价格范围: {self.df['low'].min():.4f} → "
              f"{self.df['high'].max():.4f}")

        if self.loader.issues:
            for issue in self.loader.issues:
                print(f"  {issue}")
        else:
            print("  ✔ 数据质量检查通过")

        # 构建多时间框架
        self.multi_tf = self.loader.build_multi_tf(
            ["15min", "30min", "1h", "4h", "D"]
        )
        print(f"  多频率数据: {list(self.multi_tf.keys())}")

        return self.df

    def run_phase1(self) -> List[Pivot]:
        """Phase 1：识别极值点。"""
        if self.df.empty:
            self.run_phase0()

        print("\n[Phase 1] 识别极值点...")

        # 应用硬锚点
        hard_times = self.anchor_mgr.get_hard_pivot_times()
        engine = PivotEngine(
            atr_period=self.atr_period,
            atr_mult=self.atr_mult,
            hard_pivot_times=hard_times,
        )
        self.pivots, self.atr = engine.find_pivots(self.df)

        # 应用锚点标记
        self.pivots = self.anchor_mgr.apply_to_pivots(self.pivots, self.df)

        print(f"  ✔ {len(self.pivots)} 个 pivot")
        for p in self.pivots:
            print(f"    {p}")

        # 多尺度
        self.msp = MultiScalePivots(
            self.df,
            scales=self.atr_scales,
            atr_period=self.atr_period,
        )
        print(f"\n{self.msp.summary()}")

        nesting_issues = self.msp.verify_nesting()
        if nesting_issues:
            print(f"  ⚠ 嵌套问题: {len(nesting_issues)} 个")
        else:
            print("  ✔ 嵌套验证通过")

        return self.pivots

    def run_phase2(self) -> List[ImpulseCount]:
        """Phase 2：搜索推动浪候选。"""
        if not self.pivots:
            self.run_phase1()

        print("\n[Phase 2] 搜索推动浪...")

        # 解析结构锚点
        structural_anchors = self.anchor_mgr.resolve_structural_to_pivot_indices(
            self.pivots
        )

        # 上升推动浪
        searcher_up = WaveSearcher(
            self.pivots, direction="up",
            anchors=structural_anchors, top_k=self.top_k
        )
        self.candidates_up = searcher_up.search()

        # 下降推动浪
        searcher_dn = WaveSearcher(
            self.pivots, direction="down",
            anchors=structural_anchors, top_k=self.top_k
        )
        self.candidates_dn = searcher_dn.search()

        # 调整浪
        corr_dn = CorrectionSearcher(self.pivots, direction="down", top_k=self.top_k)
        self.corrections = corr_dn.search()

        # 合并并排序
        all_cands = self.candidates_up + self.candidates_dn
        all_cands.sort(key=lambda c: c.score, reverse=True)
        self.candidates = all_cands[: self.top_k]

        print(f"  ✔ 上升推动浪候选: {len(self.candidates_up)} 个")
        print(f"  ✔ 下降推动浪候选: {len(self.candidates_dn)} 个")
        print(f"  ✔ 调整浪候选: {len(self.corrections)} 个")

        for i, c in enumerate(self.candidates):
            print(f"\n  --- 候选 #{i+1} ---")
            print("  " + c.summary().replace("\n", "\n  "))

        return self.candidates

    def run_phase3(self) -> NestedResult:
        """Phase 3：嵌套分析。"""
        if self.df.empty:
            self.run_phase0()

        print("\n[Phase 3] 嵌套分析...")

        if self.use_multi_tf:
            # 多时间框架模式
            mtf_data = {**self.multi_tf, "5min": self.df}
            analyzer = MultiTFNestedAnalyzer(
                mtf_data,
                top_k=self.top_k,
                max_depth=self.max_depth,
                direction=self.direction,
            )
        else:
            # ATR 多尺度模式
            analyzer = NestedAnalyzer(
                self.df,
                atr_scales=self.atr_scales,
                top_k=self.top_k,
                max_depth=self.max_depth,
                direction=self.direction,
                atr_period=self.atr_period,
            )

        self.nested_result = analyzer.analyze()

        print(f"  ✔ 置信度: {self.nested_result.confidence:.4f}")
        print(f"  ✔ 当前位置: {self.nested_result.current_position}")

        return self.nested_result

    def run(self):
        """一键运行 Phase 0 → 1 → 2 → 3。"""
        print("=" * 60)
        print("Elliott Wave 分析系统 — 完整运行")
        print("=" * 60)
        self.run_phase0()
        self.run_phase1()
        self.run_phase2()
        self.run_phase3()
        print("\n" + "=" * 60)
        print("分析完成")
        print("=" * 60)

    # ------------------------------------------------------------------ #
    #  报告与可视化
    # ------------------------------------------------------------------ #

    def print_report(self):
        """打印完整分析报告。"""
        print("\n" + "=" * 70)
        print("Elliott Wave 分析报告")
        print("=" * 70)

        print(f"\n文件: {self.csv_path}")
        if not self.df.empty:
            print(f"K线数: {len(self.df)}")
            print(f"时间范围: {self.df['datetime'].iloc[0]} → "
                  f"{self.df['datetime'].iloc[-1]}")

        print(f"\nPhase 1: {len(self.pivots)} 个 pivot")

        print(f"\nPhase 2: {len(self.candidates)} 个推动浪候选")
        for i, c in enumerate(self.candidates):
            print(f"\n  候选 #{i+1}:")
            print("  " + c.summary().replace("\n", "\n  "))

        if self.nested_result:
            print(f"\nPhase 3 嵌套分析:")
            self.nested_result.print_tree(max_depth=self.max_depth)

        print("\n" + "=" * 70)

    def plot(
        self,
        save_prefix: str = "ew_output",
        show: bool = True,
    ):
        """绘制所有图表。"""
        plotter = WavePlotter(self.df, self.pivots, self.candidates)

        # Phase 2 图
        plotter.plot_phase2(
            title=f"Elliott Wave Phase 0-2 — {self.csv_path.name}",
            save_path=f"{save_prefix}_phase02.png",
            show=show,
        )

        # Phase 3 嵌套图
        if self.nested_result:
            plotter.plot_nested(
                self.nested_result,
                title=f"Elliott Wave Phase 3 — {self.csv_path.name}",
                save_path=f"{save_prefix}_phase03.png",
                show=show,
                max_depth=self.max_depth,
            )

        # 多尺度图
        if self.msp:
            plotter.plot_multi_scale(
                self.msp,
                save_path=f"{save_prefix}_multiscale.png",
                show=show,
            )
````


**Phase 3的核心改进（相比之前的对话版本）：**

1. **真正的自顶向下约束传递**：`WaveConstraints`对象从父浪传递时间/价格/方向约束给子浪搜索
2. **端点对齐**：`_align_endpoints()`强制子浪起终点与父浪端点对齐（Elliott Wave嵌套铁律）
3. **双模式嵌套**：`NestedAnalyzer`（ATR多尺度）和`MultiTFNestedAnalyzer`（真实多时间框架）
4. **波浪树结构**：`WaveNode`树形结构，支持递归遍历和当前位置判断
5. **调整浪识别**：偶数子浪（2、4浪）自动用`CorrectionSearcher`识别Zigzag
